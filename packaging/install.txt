@echo off
REM ============================================================
REM voice-ime Windows 客户端一键安装脚本（v3：仅客户端）
REM 复用本机已有的 CapsWriter 服务端（Qwen3-ASR-1.7B）
REM 不装模型、不装 llama-server，安装包体积小、速度快
REM ============================================================
setlocal enabledelayedexpansion

echo.
echo  ==============================================
echo   voice-ime Windows 客户端一键安装
echo  ==============================================
echo.

REM ---------- 1. 检查 Python ----------
python --version >nul 2>&1
if errorlevel 1 (
    echo [1/3] 未检测到 Python，正在通过 winget 安装 Python 3.12 ...
    winget install -e --id Python.Python.3.12 --silent --accept-package-agreements --accept-source-agreements
    if errorlevel 1 (
        echo.
        echo [错误] Python 自动安装失败，请手动安装 Python 3.12：
        echo        https://www.python.org/downloads/ （安装时勾选 Add to PATH）
        echo        装好后重新运行本脚本。
        pause
        exit /b 1
    )
    echo [提示] Python 安装完成，请关闭本窗口后重新运行 install.bat
    pause
    exit /b 0
)
for /f "tokens=2" %%v in ('python --version 2^>^&1') do set PYVER=%%v
echo [1/3] Python %PYVER% 已安装

REM ---------- 2. 安装客户端依赖 ----------
echo.
echo [2/3] 安装客户端依赖（约需几分钟）...
python -m pip install --upgrade pip -q
python -m pip install -q -r requirements-client.txt
if errorlevel 1 (
    echo [错误] 依赖安装失败，请检查网络后重试。
    pause
    exit /b 1
)
echo        依赖安装完成

REM ---------- 3. 生成启动脚本 ----------
echo.
echo [3/3] 生成启动脚本...
(
echo @echo off
echo cd /d %%~dp0
echo echo 正在启动 voice-ime 客户端...
echo echo 请确保 CapsWriter 服务端已在运行（原版或 voice-ime 版均可）
echo python start_client.py
echo pause
) > start_client.bat

echo.
echo  ==============================================
echo   安装完成！
echo.
echo   使用方法：
echo     1. 确保 CapsWriter 服务端正在运行
echo        （你原来装的那个，不用动）
echo     2. 右键托盘退出原版 CapsWriter 客户端
echo        （避免两个客户端抢 CapsLock）
echo     3. 双击 start_client.bat
echo        （托盘出现 "voice-ime" 图标即就绪）
echo     4. 打开记事本，按住 CapsLock 说话，松开即上屏
echo.
echo   说明：本客户端直连你现有的服务端（6016 端口），
echo         复用其中的 Qwen3-ASR-1.7B 模型，不重复下载。
echo  ==============================================
echo.
pause
